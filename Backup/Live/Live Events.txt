/* =========================================================================
   X HORIZON LIVE — SPACE OPERATIONS &amp; EVENT CENTER
   (single standalone file — no build step, no server, no framework)

   Architecture summary (unchanged core, extended shell):
   - CONFIG isolates every endpoint/tunable.
   - Every raw API/snapshot record becomes ONE normalized event shape.
   - Live status is a state machine (classifyEvent), never inferred from
     &quot;scheduled time has passed&quot; alone:
       CANCELLED / SCRUBBED  (terminal, always wins)
       COMPLETED             (explicit success/failure status)
       LIVE                  (explicit webcast_live flag, or explicit
                              &quot;in flight&quot;-style status text)
       DELAYED               (explicit delayed/hold status)
       UPCOMING / STARTING   (scheduled time still ahead)
       UNKNOWN_LIVE          (scheduled time passed, nothing confirms
                              live or completed — fallback only)
       COMPLETED (fallback)  (UNKNOWN_LIVE too long with no confirmation)
   - Data has three tiers: LIVE (fresh fetch) -&gt; CACHED (last successful
     fetch) -&gt; SNAPSHOT (embedded). A failed refresh never blanks the page.
   - Rendering is now split into a persistent SHELL (header controls,
     search box, filter selects, modal skeleton — built once) and
     targeted CONTENT CONTAINERS (main stage / grid / history / stat
     numbers) that are only replaced when the thing they show actually
     changes. This means typing in search, or a background countdown
     tick, never destroys an active video iframe or steals input focus.
   - The 1s ticker only patches countdown/timer/clock text nodes unless
     the computed &quot;signature&quot; (main event + phase + data tier) changes.
   ========================================================================= */

  try {
    const res = await fetchWithTimeout(directUrl, CONFIG.REQUEST_TIMEOUT_MS);
    if (!res.ok) throw new Error('HTTP ' + res.status);
    return JSON.parse(await res.text());
  } catch (directErr) {
    if (!CONFIG.PROXY_URL) throw directErr;
    const proxied = CONFIG.PROXY_URL + encodeURIComponent(directUrl);
    const res2 = await fetchWithTimeout(proxied, CONFIG.REQUEST_TIMEOUT_MS);
    if (!res2.ok) throw new Error('HTTP ' + res2.status + ' via proxy');
    return JSON.parse(await res2.text());

function safeUrl(raw){
  if (raw == null) return null;
  let s = String(raw).trim();
  if (!s) return null;
  let m = s.match(/^\[[^\]]*\]\((https?:\/\/[^\s)]+)\)$/i);
  if (m) s = m[1];
  m = s.match(/^[\[(](https?:\/\/[^\s)\]]+)[\])]$/i);
  if (m) s = m[1];
  try {
    const u = new URL(s);
    if (u.protocol !== 'http:' &amp;&amp; u.protocol !== 'https:') return null;
    return u.href;
  } catch (e) { return null; }
}

function diffParts(targetMs, nowMs){
  let d = Math.max(0, targetMs - nowMs);
  const days = Math.floor(d / 86400000); d -= days * 86400000;
  const hours = Math.floor(d / 3600000); d -= hours * 3600000;
  const minutes = Math.floor(d / 60000); d -= minutes * 60000;
  const seconds = Math.floor(d / 1000);
  return { days, hours, minutes, seconds };
}

  const seconds = Math.floor(d / 1000);
  return { hours, minutes, seconds };
}

// Local-calendar-day bucket relative to &quot;now&quot; — used by the date-range tabs.
function dateBucket(scheduledTime, nowMs){
  const t = new Date(scheduledTime).getTime();
  if (isNaN(t)) return 'INVALID';
  const d = new Date(t), n = new Date(nowMs);
  const dMid = new Date(d.getFullYear(), d.getMonth(), d.getDate()).getTime();
  const nMid = new Date(n.getFullYear(), n.getMonth(), n.getDate()).getTime();
  const dayDiff = Math.round((dMid - nMid) / 86400000);
  if (dayDiff &lt; 0) return 'PAST';
  if (dayDiff === 0) return 'TODAY';
  if (dayDiff === 1) return 'TOMORROW';
  if (dayDiff &lt;= 6) return 'NEXT7';
  return 'LATER';
}

function classifyStatus(text){
  if (!text) return 'UNKNOWN';
  const s = String(text);
  if (/cancel/i.test(s)) return 'CANCELLED';
  if (/scrub/i.test(s)) return 'SCRUBBED';
  if (/partial failure/i.test(s)) return 'FAILURE';
  if (/failure/i.test(s)) return 'FAILURE';
  if (/success/i.test(s)) return 'SUCCESS';
  if (/in[\s-]?flight|ongoing|underway/i.test(s)) return 'IN_FLIGHT';
  if (/^live$/i.test(s)) return 'IN_FLIGHT';
  if (/delay/i.test(s)) return 'DELAYED';
  if (/hold/i.test(s)) return 'HOLD';
  if (/go for launch/i.test(s) || /^go$/i.test(s)) return 'GO';
  if (/to be determined/i.test(s) || /^tbd$/i.test(s)) return 'TBD';
  return 'UNKNOWN';
}

    let statusCategory = classifyStatus(statusName);

    return {
      id: 'launch-' + l.id,
      type: 'launch',
      name: String(l.name).trim(),

      location: (l.pad &amp;&amp; ((l.pad.location &amp;&amp; l.pad.location.name) || l.pad.name)) || null,
      status: statusName,
      statusCategory: statusCategory,
      webcastUrl: webcastUrls[0] || null,

      webcastLive: typeof l.webcast_live === 'boolean' ? l.webcast_live : null,
      hasWebcast: webcastUrls.length &gt; 0,

      hasWebcast: webcastUrls.length &gt; 0,
      missionType: tn || null,
      orbit: null,
      rocketName: null,
      sourceUrl: safeUrl(e.url)
    };
  } catch (e2) { return null; }
}

function normalizeSnapshotItem(s){
  try {
    if (!s || !s.id || !s.name || !s.net || isNaN(new Date(s.net).getTime())) return null;
    const webcastUrls = (s.vid_urls || []).map(safeUrl).filter(Boolean);
    const rawId = String(s.id).replace(/^(launch|event)-/, '');
    const sourceUrl = safeUrl(CONFIG.API_BASE + '/' + (s.type === 'event' ? 'events' : 'launches') + '/' + rawId + '/');
    return {
      id: s.id,
      type: s.type === 'event' ? 'event' : 'launch',
      name: String(s.name).trim(),
      description: trim(s.description || ''),
      scheduledTime: s.net,
      imageUrl: safeUrl(s.image_url),
      agency: s.agency || null,
      location: s.location || null,
      status: s.status || null,
      statusCategory: classifyStatus(s.status),
      webcastUrl: webcastUrls[0] || null,
      webcastUrls: webcastUrls,
      webcastLive: null,
      hasWebcast: webcastUrls.length &gt; 0,
      missionType: s.mission_type || null,
      orbit: s.orbit || null,
      rocketName: s.rocket_name || null,
      sourceUrl: sourceUrl
    };
  } catch (e) { return null; }
}

function dedupe(list){
  const seen = new Map();
  for (const ev of list) { if (ev &amp;&amp; ev.id &amp;&amp; !seen.has(ev.id)) seen.set(ev.id, ev); }
  return Array.from(seen.values());
}

function sourceHost(url){
  try { return new URL(url).hostname.replace(/^www\./,'').toLowerCase(); } catch(e) { return ''; }

function isOfficialSource(url){
  const h = sourceHost(url);
  return !!h &amp;&amp; OFFICIAL_HOSTS.some(x =&gt; h === x || h.endsWith('.' + x));

function sourceLabel(url, index){
  const h = sourceHost(url);

  if (/x\.com|twitter\.com/i.test(h)) return 'X';
  return h ? h.replace(/^www\./,'').split('.')[0] : 'Source ' + (index + 1);
}

function classifyEvent(ev, nowMs){
  const cat = ev.statusCategory;
  if (cat === 'CANCELLED') return { phase: 'CANCELLED' };
  if (cat === 'SCRUBBED') return { phase: 'SCRUBBED' };
  if (cat === 'SUCCESS' || cat === 'FAILURE') return { phase: 'COMPLETED', confirmed: true };

  const confirmedLive = ev.webcastLive === true || cat === 'IN_FLIGHT';
  if (confirmedLive) return { phase: 'LIVE', confirmed: true };

  if (cat === 'DELAYED' || cat === 'HOLD') return { phase: 'DELAYED' };

  const schedMs = new Date(ev.scheduledTime).getTime();
  if (isNaN(schedMs)) return { phase: 'UNKNOWN' };
  const delta = nowMs - schedMs;

  if (delta &lt; -CONFIG.STARTING_WINDOW_MS) return { phase: 'UPCOMING' };
  if (delta &lt; 0) return { phase: 'STARTING' };

  // Once the scheduled start has passed, a linked webcast is treated as the
  // active coverage window. This prevents the old &quot;Status Unknown&quot; dead zone
  // and keeps the Live counter aligned with the main stage.
  if (delta &lt;= CONFIG.UNCONFIRMED_LIVE_WINDOW_MS &amp;&amp; ev.hasWebcast) {
    return { phase: 'LIVE', inferred: true };
  }
  if (delta &lt;= CONFIG.UNCONFIRMED_LIVE_WINDOW_MS) return { phase: 'UNKNOWN_LIVE' };
  return { phase: 'COMPLETED', confirmed: false };
}

function selectMainStage(classified){
  const live = classified.filter(c =&gt; c.phase === 'LIVE');
  if (live.length){
    live.sort((a, b) =&gt; {
      const aw = a.ev.hasWebcast ? 0 : 1, bw = b.ev.hasWebcast ? 0 : 1;
      if (aw !== bw) return aw - bw;
      return new Date(a.ev.scheduledTime) - new Date(b.ev.scheduledTime);
    });
    return live[0];
  }
  const unknownLive = classified.filter(c =&gt; c.phase === 'UNKNOWN_LIVE');
  if (unknownLive.length){
    unknownLive.sort((a, b) =&gt; new Date(b.ev.scheduledTime) - new Date(a.ev.scheduledTime));
    return unknownLive[0];
  }
  const upcoming = classified.filter(c =&gt; c.phase === 'UPCOMING' || c.phase === 'STARTING' || c.phase === 'DELAYED');
  if (upcoming.length){
    upcoming.sort((a, b) =&gt; new Date(a.ev.scheduledTime) - new Date(b.ev.scheduledTime));
    return upcoming[0];
  }
  return null;
}

function phaseLabel(phase){
  switch (phase){
    case 'LIVE': return 'Live Now';
    case 'UNKNOWN_LIVE': return 'Live Coverage';
    case 'STARTING': return 'Starting Soon';
    case 'UPCOMING': return 'Upcoming';
    case 'DELAYED': return 'Delayed';
    case 'COMPLETED': return 'Completed';
    case 'CANCELLED': return 'Cancelled';
    case 'SCRUBBED': return 'Scrubbed';
    default: return 'Unknown';
  }
}

function computeCounts(classified){
  const counts = { LIVE: 0, STARTING: 0, UPCOMING: 0, DELAYED: 0, UNKNOWN_LIVE: 0, COMPLETED: 0, CANCELLED: 0, TOTAL: classified.length };
  for (const c of classified){
    if (c.phase === 'CANCELLED' || c.phase === 'SCRUBBED') counts.CANCELLED++;
    else if (counts[c.phase] !== undefined) counts[c.phase]++;
  }
  return counts;
}

function matchesSearch(ev, q){
  if (!q) return true;
  const hay = [ev.name, ev.agency, ev.location, ev.rocketName, ev.missionType, ev.orbit, ev.status].filter(Boolean).join(' ').toLowerCase();
  return hay.indexOf(q.toLowerCase()) !== -1;
}

function matchesFilters(item, uiState, nowMs){
  const ev = item.ev, phase = item.phase;
  if (uiState.statusFilter !== 'ALL'){
    if (uiState.statusFilter === 'CANCELLED'){ if (phase !== 'CANCELLED' &amp;&amp; phase !== 'SCRUBBED') return false; }
    else if (phase !== uiState.statusFilter) return false;
  }
  if (uiState.typeFilter !== 'ALL' &amp;&amp; ev.type !== uiState.typeFilter) return false;
  if (uiState.webcastFilter === 'YES' &amp;&amp; !ev.hasWebcast) return false;
  if (uiState.webcastFilter === 'NO' &amp;&amp; ev.hasWebcast) return false;
  if (uiState.providerFilter !== 'ALL' &amp;&amp; ev.agency !== uiState.providerFilter) return false;
  if (uiState.dateView !== 'ALL' &amp;&amp; dateBucket(ev.scheduledTime, nowMs) !== uiState.dateView) return false;
  if (!matchesSearch(ev, uiState.search)) return false;
  return true;
}

function computeFilteredGrid(classified, uiState, nowMs){
  return classified.filter(item =&gt; matchesFilters(item, uiState, nowMs))
    .sort((a, b) =&gt; new Date(a.ev.scheduledTime) - new Date(b.ev.scheduledTime));
}

function computeHistory(classified, limit){
  return classified
    .filter(c =&gt; c.phase === 'COMPLETED' || c.phase === 'CANCELLED' || c.phase === 'SCRUBBED')
    .sort((a, b) =&gt; new Date(b.ev.scheduledTime) - new Date(a.ev.scheduledTime))
    .slice(0, limit || 24);
}

let prevPhaseMap = new Map();
function detectTransitions(classified){
  const transitions = [];
  const nextMap = new Map();
  for (const c of classified){
    nextMap.set(c.ev.id, c.phase);
    const prev = prevPhaseMap.get(c.ev.id);
    if (prev === undefined) transitions.push({ id: c.ev.id, from: null, to: c.phase, isNew: true });
    else if (prev !== c.phase) transitions.push({ id: c.ev.id, from: prev, to: c.phase, isNew: false });
  }
  prevPhaseMap = nextMap;
  return transitions;
}

function fetchWithTimeout(url, ms){
  const controller = new AbortController();
  const timer = setTimeout(() =&gt; controller.abort(), ms);
  return fetch(url, { signal: controller.signal }).finally(() =&gt; clearTimeout(timer));
}

async function fetchEvents(){
  if (refreshing) return;
  refreshing = true;
  updateHeaderSync();
  try {
    const [ld, ed] = await Promise.all([
      fetchJson(CONFIG.LAUNCHES_PATH),
      fetchJson(CONFIG.EVENTS_PATH)
    ]);
    const launches = (ld.results || []).map(normalizeLaunch).filter(Boolean);
    const evs = (ed.results || []).map(normalizeEvent).filter(Boolean);
    const merged = dedupe([...launches, ...evs]);
    if (!merged.length) throw new Error('No valid events in response');
    events = merged;
    lastGoodEvents = merged;
    dataState = 'live';
    lastSuccessAt = new Date();
    lastFetchFailed = false;
  } catch (err) {
    lastFetchFailed = true;
    if (lastGoodEvents) { events = lastGoodEvents; dataState = 'cached'; }
    else { dataState = 'snapshot'; }
  }
  refreshing = false;
  guardedFullRender();
}

    } else if (stage.requestFullscreen) {
      await stage.requestFullscreen();
    }
  } catch (err) {
    console.warn('X HORIZON fullscreen unavailable', err);
  }
  updateMainStageFullscreenButton();
}

    if (el) openModal(el.getAttribute('data-open-id'));
  });

  if (modalCloseBtn) modalCloseBtn.addEventListener('click', closeModal);
  const modalOverlay = document.getElementById('modalOverlay');
  if (modalOverlay) modalOverlay.addEventListener('click', e =&gt; { if (e.target.id === 'modalOverlay') closeModal(); });

function mainCountdownHTML(scheduledTime, nowMs){
  const { days, hours, minutes, seconds } = diffParts(new Date(scheduledTime).getTime(), nowMs);
  if (days === 0 &amp;&amp; hours === 0 &amp;&amp; minutes === 0 &amp;&amp; seconds === 0){
    return '&lt;div class=&quot;countdown&quot; id=&quot;mainCountdown&quot;&gt;&lt;div class=&quot;cd-block&quot; style=&quot;min-width:96px&quot;&gt;&lt;span class=&quot;v&quot;&gt;T-0&lt;/span&gt;&lt;/div&gt;&lt;/div&gt;';
  }
  const blocks = [['Days', 'mainCdDays', days], ['Hrs', 'mainCdHrs', hours], ['Min', 'mainCdMin', minutes], ['Sec', 'mainCdSec', seconds]];
  return '&lt;div class=&quot;countdown&quot; id=&quot;mainCountdown&quot;&gt;' + blocks.map(b =&gt; '&lt;div class=&quot;cd-block&quot;&gt;&lt;span class=&quot;v&quot; id=&quot;' + b[1] + '&quot;&gt;' + pad2(b[2]) + '&lt;/span&gt;&lt;span class=&quot;l&quot;&gt;' + b[0] + '&lt;/span&gt;&lt;/div&gt;').join('') + '&lt;/div&gt;';
}

function compactCountdownHTML(ev, nowMs){
  const { days, hours, minutes, seconds } = diffParts(new Date(ev.scheduledTime).getTime(), nowMs);
  const label = (days === 0 &amp;&amp; hours === 0 &amp;&amp; minutes === 0 &amp;&amp; seconds === 0) ? 'T-0' : ((days &gt; 0 ? days + 'd ' : '') + pad2(hours) + ':' + pad2(minutes) + ':' + pad2(seconds));
  return '&lt;span class=&quot;countdown-compact&quot; data-cid=&quot;' + esc(ev.id) + '&quot;&gt;' + label + '&lt;/span&gt;';
}

  const sel = document.getElementById('providerFilterSel');
  if (!sel) return;

  sel.innerHTML = '&lt;option value=&quot;ALL&quot;&gt;All Providers&lt;/option&gt;' + list.map(p =&gt; '&lt;option value=&quot;' + esc(p) + '&quot;&gt;' + esc(p) + '&lt;/option&gt;').join('');

  else { sel.value = 'ALL'; uiState.providerFilter = 'ALL'; }
}

    ? '&lt;button type=&quot;button&quot; class=&quot;btn&quot; data-modal-source=&quot;' + esc(s.url) + '&quot;&gt;' + IC.play + ' ' + esc(s.label) + '&lt;/button&gt;'

  ).join('') + '&lt;/div&gt;' : '';

  const rows = [];
  rows.push(['Status', esc(phaseLabel(phase)) + (ev.status ? ' &lt;span class=&quot;status-pill&quot; style=&quot;margin-left:6px&quot;&gt;' + esc(ev.status) + '&lt;/span&gt;' : '')]);
  if (phase === 'LIVE') rows.push(['Elapsed', '&lt;span id=&quot;modalTPlus&quot;&gt;T+ 00:00:00&lt;/span&gt;']);
  rows.push(['Scheduled (UTC)', esc(formatUtc(ev.scheduledTime))]);
  rows.push(['Scheduled (Local)', esc(formatLocalFull(ev.scheduledTime))]);
  if (ev.missionType) rows.push(['Mission Type', esc(ev.missionType)]);
  if (ev.agency) rows.push(['Provider', esc(ev.agency)]);
  if (ev.location) rows.push(['Location', esc(ev.location)]);
  if (ev.rocketName) rows.push(['Rocket', esc(ev.rocketName)]);
  if (ev.orbit) rows.push(['Orbit', esc(ev.orbit)]);

  const rowsHTML = rows.map(r =&gt; '&lt;div class=&quot;modal-row&quot;&gt;&lt;span class=&quot;modal-row-label&quot;&gt;' + r[0] + '&lt;/span&gt;&lt;span class=&quot;modal-row-value&quot;&gt;' + r[1] + '&lt;/span&gt;&lt;/div&gt;').join('');
  const tagCls = ev.type === 'launch' ? 'tag tag-launch' : 'tag tag-event';
  const tagIcon = ev.type === 'launch' ? IC.rocket : IC.radio;

  return '&lt;h2 id=&quot;modalTitleText&quot;&gt;' + esc(ev.name) + '&lt;/h2&gt;' +
    '&lt;div class=&quot;tags&quot; style=&quot;margin-top:8px&quot;&gt;&lt;span class=&quot;' + tagCls + '&quot;&gt;' + tagIcon + esc(ev.type) + '&lt;/span&gt;&lt;/div&gt;' +
    '&lt;div class=&quot;modal-rows&quot;&gt;' + rowsHTML + '&lt;/div&gt;' +
    (ev.description ? '&lt;p class=&quot;modal-desc&quot;&gt;' + esc(ev.description) + '&lt;/p&gt;' : '') +
    '&lt;div class=&quot;modal-webcast&quot;&gt;' + webcastBlock + '&lt;/div&gt;' + sourceRows +
    '&lt;div class=&quot;modal-actions&quot;&gt;&lt;button type=&quot;button&quot; class=&quot;btn&quot; id=&quot;modalShareBtn&quot;&gt;' + IC.external + ' Share&lt;/button&gt;' +
    (ev.sourceUrl ? '&lt;a class=&quot;btn&quot; href=&quot;' + esc(ev.sourceUrl) + '&quot; target=&quot;_blank&quot; rel=&quot;noopener noreferrer&quot;&gt;Source Data&lt;/a&gt;' : '') +
    '&lt;/div&gt;';
}

async function shareEvent(ev, phase){
  const text = ev.name + ' \u2014 ' + phaseLabel(phase) + ' \u2014 ' + formatUtc(ev.scheduledTime);
  const shareData = { title: 'X HORIZON: ' + ev.name, text: text };
  if (ev.sourceUrl) shareData.url = ev.sourceUrl;
  try {
    if (navigator.share) { await navigator.share(shareData); return; }
  } catch (e) { /* user cancelled or share failed; fall through to clipboard */ }
  try {
    const clip = text + (ev.sourceUrl ? ('\n' + ev.sourceUrl) : '');
    await navigator.clipboard.writeText(clip);
    const btn = document.getElementById('modalShareBtn');
    if (btn) { const orig = btn.textContent; btn.textContent = 'Copied!'; setTimeout(() =&gt; { btn.textContent = orig; }, 1500); }
  } catch (e2) { /* clipboard unavailable — nothing more we can safely do */ }
}

function bindModalShareButton(ev, phase){
  const shareBtn = document.getElementById('modalShareBtn');
  if (shareBtn) shareBtn.addEventListener('click', () =&gt; shareEvent(ev, phase));
}

function openModal(id){
  const item = lastClassified.find(c =&gt; c.ev.id === id);
  if (!item) return;
  openModalId = id;
  lastFocusedBeforeModal = document.activeElement;
  setHtml('modalBody', modalBodyHTML(item.ev, item.phase));
  bindModalShareButton(item.ev, item.phase);
  const overlay = document.getElementById('modalOverlay');
  if (overlay) overlay.hidden = false;
  const closeBtn = document.getElementById('modalCloseBtn');
  if (closeBtn) closeBtn.focus();
  document.addEventListener('keydown', onModalKeydown);
}

function closeModal(){
  const overlay = document.getElementById('modalOverlay');
  if (overlay) overlay.hidden = true;
  openModalId = null;
  document.removeEventListener('keydown', onModalKeydown);
  if (lastFocusedBeforeModal &amp;&amp; typeof lastFocusedBeforeModal.focus === 'function') lastFocusedBeforeModal.focus();
}

function refreshModalIfOpen(){
  if (!openModalId) return;
  const item = lastClassified.find(c =&gt; c.ev.id === openModalId);
  if (!item) { closeModal(); return; }
  setHtml('modalBody', modalBodyHTML(item.ev, item.phase));
  bindModalShareButton(item.ev, item.phase);
}

function onModalKeydown(e){
  if (e.key === 'Escape') { closeModal(); return; }
  if (e.key === 'Tab'){
    const panel = document.getElementById('modalPanel');
    if (!panel) return;
    const focusables = panel.querySelectorAll('a[href], button:not([disabled]), input, select, textarea, [tabindex]:not([tabindex=&quot;-1&quot;])');
    if (!focusables.length) return;
    const first = focusables[0], last = focusables[focusables.length - 1];
    if (e.shiftKey &amp;&amp; document.activeElement === first) { e.preventDefault(); last.focus(); }
    else if (!e.shiftKey &amp;&amp; document.activeElement === last) { e.preventDefault(); first.focus(); }
  }
}

function recomputeCore(){
  lastClassified = events.map(ev =&gt; Object.assign({ ev: ev }, classifyEvent(ev, now)));
  lastMain = selectMainStage(lastClassified);
  lastCounts = computeCounts(lastClassified);
  const transitions = detectTransitions(lastClassified);
  if (transitions.length) handleEventTransitions(transitions);
  const m = lastMain ? [lastMain.ev.id, lastMain.phase, lastMain.ev.webcastLive, lastMain.ev.webcastUrl, (lastMain.ev.webcastUrls||[]).join('|')].join('~') : 'none';
  const c = lastClassified.map(x =&gt; x.ev.id + ':' + x.phase).join(',');
  return [m, c, dataState, lastFetchFailed, refreshing].join('||');
}

function renderResults(){
  renderStatOverview(lastCounts);
  renderMainStage(lastMain);
  renderGrid(computeFilteredGrid(lastClassified, uiState, now));
  renderHistory(computeHistory(lastClassified, CONFIG.HISTORY_LIMIT));
  updateProviderFilterOptions(lastClassified);
  refreshModalIfOpen();
}

function guardedRenderResults(){
  try { renderResults(); } catch (err) { console.error('X HORIZON render error', err); }
}

function updateLiveNumbers(){
  const nowMs = now;
  if (lastMain){
    const ev = lastMain.ev, phase = lastMain.phase;
    const schedMs = new Date(ev.scheduledTime).getTime();
    if (phase === 'UPCOMING' || phase === 'STARTING' || (phase === 'DELAYED' &amp;&amp; schedMs &gt; nowMs)){
      const { days, hours, minutes, seconds } = diffParts(schedMs, nowMs);
      if (document.getElementById('mainCdDays')){
        setText('mainCdDays', pad2(days)); setText('mainCdHrs', pad2(hours)); setText('mainCdMin', pad2(minutes)); setText('mainCdSec', pad2(seconds));
        if (days === 0 &amp;&amp; hours === 0 &amp;&amp; minutes === 0 &amp;&amp; seconds === 0){
          const cd = document.getElementById('mainCountdown');
          if (cd) cd.innerHTML = '&lt;div class=&quot;cd-block&quot; style=&quot;min-width:96px&quot;&gt;&lt;span class=&quot;v&quot;&gt;T-0&lt;/span&gt;&lt;/div&gt;';
        }
      }
    } else if (phase === 'LIVE'){
      const { hours, minutes, seconds } = elapsedParts(schedMs, nowMs);
      setText('mainTPlus', 'T+ ' + pad2(hours) + ':' + pad2(minutes) + ':' + pad2(seconds));
    }
  }
  document.querySelectorAll('[data-cid]').forEach(node =&gt; {
    const id = node.getAttribute('data-cid');
    const item = lastClassified.find(c =&gt; c.ev.id === id);
    if (!item) return;
    const { days, hours, minutes, seconds } = diffParts(new Date(item.ev.scheduledTime).getTime(), nowMs);
    node.textContent = (days === 0 &amp;&amp; hours === 0 &amp;&amp; minutes === 0 &amp;&amp; seconds === 0) ? 'T-0' : ((days &gt; 0 ? days + 'd ' : '') + pad2(hours) + ':' + pad2(minutes) + ':' + pad2(seconds));
  });
  if (openModalId &amp;&amp; lastMain &amp;&amp; lastMain.ev.id === openModalId &amp;&amp; lastMain.phase === 'LIVE'){
    const { hours, minutes, seconds } = elapsedParts(new Date(lastMain.ev.scheduledTime).getTime(), nowMs);
    setText('modalTPlus', 'T+ ' + pad2(hours) + ':' + pad2(minutes) + ':' + pad2(seconds));
  }
}

function ensureShell(){ if (!shellBuilt) { buildShell(); shellBuilt = true; } }

function fullRender(){
  ensureShell();
  lastSignature = recomputeCore();
  updateHeaderSync();
  updateClocks();
  updateConnDot();
  renderResults();
}

function guardedFullRender(){
  try { fullRender(); }
  catch (err){
    console.error('X HORIZON fatal render error', err);
    const app = document.getElementById('app');
    if (app){
      app.innerHTML = '&lt;div class=&quot;empty&quot;&gt;Something went wrong displaying live events.&lt;br&gt;&lt;button class=&quot;btn&quot; id=&quot;reloadBtn&quot; style=&quot;margin-top:16px&quot;&gt;Reload page&lt;/button&gt;&lt;/div&gt;';
      const rb = document.getElementById('reloadBtn');
      if (rb) rb.addEventListener('click', () =&gt; location.reload());
    }
  }
}

function tick(){
  now = Date.now();
  try {
    const sig = recomputeCore();
    updateClocks();
    if (sig !== lastSignature){
      lastSignature = sig;
      renderResults();
    } else {
      updateLiveNumbers();
    }
  } catch (err){ console.error('X HORIZON tick error', err); }
}
