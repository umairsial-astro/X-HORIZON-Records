  function cleanLines(value){
    if(Array.isArray(value)) return value.map(v => String(v).trim()).filter(Boolean);
    return String(value || '').split(/\r?\n|;/).map(v => v.trim()).filter(Boolean);
  }

  function normalizeCompany(company, index){
    const c = Object.assign({}, company);
    c.roles = cleanLines(c.roles);
    c.req = cleanLines(c.req);
    c.steps = cleanLines(c.steps);
    c.missions = cleanLines(getMissions(c));
    c.founded = String(c.founded || c.foundingDate || c.established || foundingDates[c.name] || '').trim();
    c.primaryLabel = String(c.primaryLabel || c.button1Label || c.popupButton1Label || 'Career page').trim();
    c.primaryUrl = String(c.primaryUrl || c.button1Url || c.popupButton1Url || c.career || '#').trim();

    c.secondaryUrl = String(c.secondaryUrl || c.button2Url || c.popupButton2Url || '').trim();
    c.fallbackImage = spaceArtwork(c, index || 0);
    if(!c.image) c.image = img.launch;
    return c;
  }

  let xhAdminBridgePromise = null;

  async function ensureFirebaseAdminBridge(){
    if(window.XHorizonAdmin && typeof window.XHorizonAdmin.unlock === 'function') return window.XHorizonAdmin;
    const config = firebaseConfig();
    if(!validFirebaseConfig(config)) return null;
    if(xhAdminBridgePromise) return xhAdminBridgePromise;
    xhAdminBridgePromise = (async function(){
      const okApp = window.firebase ? true : await loadScript('https://www.gstatic.com/firebasejs/10.12.5/firebase-app-compat.js');
      const okAuth = window.firebase && window.firebase.auth ? true : await loadScript('https://www.gstatic.com/firebasejs/10.12.5/firebase-auth-compat.js');
      const okFunctions = window.firebase && window.firebase.functions ? true : await loadScript('https://www.gstatic.com/firebasejs/10.12.5/firebase-functions-compat.js');
      if(!okApp || !okAuth || !okFunctions || !window.firebase || !window.firebase.auth || !window.firebase.functions) return null;
      try{
        const app = window.firebase.apps && window.firebase.apps.length ? window.firebase.app() : window.firebase.initializeApp(config);
        const auth = window.firebase.auth(app);
        const cfg = rootConfig();
        const region = cfg.functionsRegion || cfg.firebaseFunctionsRegion || cfg.region;
        const functions = region ? window.firebase.app().functions(region) : window.firebase.functions(app);
        const fnName = cfg.verifyAdminFunction || cfg.adminFunction || cfg.verifyFunction || 'verifyAdminKey';
        window.XHorizonAdmin = {
          async unlock(key, done){
            const cleanKey = String(key || '').trim();
            if(!cleanKey) return false;
            try{
              const verifyAdmin = functions.httpsCallable(fnName);
              const result = await verifyAdmin({ key: cleanKey, page: 'world-space-companies', scope: 'space-companies' });
              const data = result && result.data ? result.data : {};
              const customToken = data.token || data.customToken || data.firebaseToken || data.authToken || '';
              if(customToken) await auth.signInWithCustomToken(customToken);
              let allowed = data.allowed === true || data.ok === true || data.admin === true || data.xhAdmin === true;
              const user = auth.currentUser;
              if(user){
                const tokenResult = await user.getIdTokenResult(true);
                const claims = tokenResult && tokenResult.claims ? tokenResult.claims : {};
                allowed = allowed || claims.admin === true || claims.xhAdmin === true || claims.xhorizonAdmin === true;
              }
              if(allowed){
                document.body.classList.add('editor-on');
                if(typeof done === 'function') done();
                return true;
              }
              return false;
            }catch(error){
              console.error('X Horizon Firebase unlock failed:', error);
              return false;
            }
          }
        };
        return window.XHorizonAdmin;
      }catch(error){
        console.error('X Horizon Firebase bridge failed:', error);
        return null;
      }
    })();
    return xhAdminBridgePromise;
  }

  function tryLocalAdminUnlock(value){
    const candidates = adminCandidates(value);
    if(!candidates.length) return false;
    const keys = adminKeys();
    if(keys.length && candidates.some(candidate => keys.includes(candidate))){
      unlockAdmin();
      if(search) search.value = '';
      render();
      return true;
    }
    return false;
  }

  async function tryAdminUnlock(value){
    if(tryLocalAdminUnlock(value)) return true;
    const candidates = adminCandidates(value);
    if(!candidates.length) return false;
    await loadOptionalRootConfig();
    if(tryLocalAdminUnlock(value)) return true;
    const bridge = await ensureFirebaseAdminBridge();
    if(bridge && typeof bridge.unlock === 'function'){
      setAdminStatus('Checking access...');
      for(const candidate of candidates){
        const ok = await Promise.resolve(bridge.unlock(candidate, function(){
          unlockAdmin();
          if(search) search.value = '';
          render();
          setAdminStatus(xhFirebaseConnected ? 'Editor unlocked — Firebase connected' : 'Editor unlocked');
        }));
        if(ok){
          if(!adminActive) unlockAdmin();
          if(search) search.value = '';
          render();
          return true;
        }
      }
    }
    return false;
  }

  async function saveAllData(remote){
    persistLocal();
    if(remote && xhFirebaseRef && adminActive){
      try{
        await xhFirebaseRef.set(companyData);
        setAdminStatus('Saved to Firebase and local storage');
      }catch(error){
        console.error('X Horizon Firebase save failed:', error);
        setAdminStatus('Saved locally; Firebase write is restricted');
      }
    }else{
      setAdminStatus(xhFirebaseConnected ? 'Saved locally; Firebase write is restricted' : 'Saved to local storage');
    }
  }

  async function loadScript(src){
    return new Promise(resolve => {
      if(document.querySelector(`script[src="${src}"]`)) return resolve(true);
      const s = document.createElement('script');
      s.src = src;
      s.async = true;
      s.onload = () => resolve(true);
      s.onerror = () => resolve(false);
      document.head.appendChild(s);
    });
  }

  async function loadOptionalRootConfig(){
    if(firebaseConfig() || adminKeys().length) return;
    const paths = ['/xh-root-config.js','/firebase-config.js','/xhorizon-root-config.js','/root-config.js','./xh-root-config.js','./firebase-config.js','./xhorizon-root-config.js','./root-config.js'];
    for(const path of paths){
      await loadScript(path);
      if(firebaseConfig() || adminKeys().length) break;
    }
  }

  async function connectFirebase(){
    await loadOptionalRootConfig();
    const config = firebaseConfig();
    if(!validFirebaseConfig(config)){
      setAdminStatus('Local editor ready');
      if(rootConfig().adminOpen === true || window.XH_ADMIN_OPEN === true) unlockAdmin();
      return;
    }
    await ensureFirebaseAdminBridge();
    const okApp = window.firebase ? true : await loadScript('https://www.gstatic.com/firebasejs/10.12.5/firebase-app-compat.js');
    const okDb = window.firebase && window.firebase.database ? true : await loadScript('https://www.gstatic.com/firebasejs/10.12.5/firebase-database-compat.js');
    if(!okApp || !okDb || !window.firebase || !window.firebase.database){
      setAdminStatus('Local editor ready');
      return;
    }
    try{
      const app = window.firebase.apps && window.firebase.apps.length ? window.firebase.app() : window.firebase.initializeApp(config);
      xhFirebaseRef = window.firebase.database(app).ref(firebaseDataPath());
      xhFirebaseRef.on('value', snapshot => {
        const value = snapshot.val();
        if(Array.isArray(value)){
          setCompanyData(value);
          persistLocal();
          render();
          if(adminActive){
            populateAdminSelect();
            loadAdminForm(Number(adminSelect.value || 0));
          }
        }
      });
      xhFirebaseConnected = true;
      setAdminStatus(adminActive ? 'Editor unlocked — Firebase connected' : 'Firebase connected');
      if(rootConfig().adminOpen === true || window.XH_ADMIN_OPEN === true) unlockAdmin();
    }catch(err){
      setAdminStatus('Local editor ready');
    }
  }

  if(adminSelect){
    adminSelect.addEventListener('change', () => loadAdminForm(Number(adminSelect.value)));
  }

  const adminSave = document.getElementById('xhAdminSave');
  const adminNew = document.getElementById('xhAdminNew');
  const adminDelete = document.getElementById('xhAdminDelete');
  const adminReset = document.getElementById('xhAdminReset');
  const adminClearAll = document.getElementById('xhAdminClearAll');
  const adminExport = document.getElementById('xhAdminExport');

  if(adminSave){
    adminSave.addEventListener('click', async () => {
      if(!adminActive) return;
      const index = Number(adminSelect.value);
      const entry = readAdminForm();
      let savedIndex = index;
      if(Number.isInteger(index) && index >= 0 && index < companyData.length){
        companyData[index] = entry;
      }else{
        companyData.push(entry);
        savedIndex = companyData.length - 1;
      }
      setCompanyData(companyData);
      render();
      populateAdminSelect();
      adminSelect.value = String(Math.max(0, savedIndex));
      loadAdminForm(Number(adminSelect.value));
      await saveAllData(true);
    });
  }

  if(adminNew){
    adminNew.addEventListener('click', () => {

      setAdminStatus('New entry ready');
    });
  }

      if(index < 0 || index >= companyData.length) return;
      companyData.splice(index, 1);
      setCompanyData(companyData);
      render();
      populateAdminSelect();
      if(companyData.length){
        loadAdminForm(Math.min(index, companyData.length - 1));
      }else{
        clearAdminFormForNew();
        setAdminStatus('All entries removed. Add a new card when ready.');

    });

      render();

    });

  if(adminReset){

      render();

    });

      link.href = URL.createObjectURL(blob);
      link.download = 'xh-space-companies-data.json';
      link.click();
      URL.revokeObjectURL(link.href);
      setAdminStatus('Exported JSON');
    });
  }

  async function handleSearchAction(){
    const value = search ? search.value.trim() : '';
    if(!value){ render(); return; }
    const oldLabel = searchButton ? searchButton.textContent : '';
    if(searchButton){ searchButton.disabled = true; searchButton.textContent = 'Checking'; }
    try{
      if(await tryAdminUnlock(value)) return;
    }catch(error){
      console.error('X Horizon unlock failed:', error);
    }finally{
      if(searchButton){ searchButton.disabled = false; searchButton.textContent = oldLabel || 'Search'; }
    }
    render();
  }

  if(search){
    search.addEventListener('input', () => {
      if(maybeAdminIntent(search.value) && tryLocalAdminUnlock(search.value)) return;
      render();
    });
    search.addEventListener('keydown', e => { if(e.key === 'Enter'){ e.preventDefault(); handleSearchAction(); } });
  }
  if(searchButton){
    searchButton.addEventListener('click', handleSearchAction);
  }

    const secondaryUrl = copyPlan.dataset.url || currentCompany.secondaryUrl || '';

      window.open(secondaryUrl, '_blank', 'noopener');
      return;
    }
    const plan = `${currentCompany.name}\nFounded / established: ${currentCompany.founded || 'Not added'}\n\nMissions / programs:\n- ${getMissions(currentCompany).join('\n- ')}\n\nRoles:\n- ${currentCompany.roles.join('\n- ')}\n\nRequirements:\n- ${currentCompany.req.join('\n- ')}\n\nHow to join:\n${currentCompany.steps.map((s,i)=>`${i+1}. ${s}`).join('\n')}\n\nCareer page: ${currentCompany.primaryUrl || currentCompany.career}`;
    const originalLabel = currentCompany.secondaryLabel || 'Copy joining plan';
    try{
      await navigator.clipboard.writeText(plan);
      copyPlan.textContent = 'Copied';
      setTimeout(()=>copyPlan.textContent=originalLabel,1200);
    }catch(err){
      copyPlan.textContent = 'Select text manually';
      setTimeout(()=>copyPlan.textContent=originalLabel,1400);
    }
  });
