async function sha256(value){
  const data = new TextEncoder().encode(value);
  const buf = await crypto.subtle.digest("SHA-256", data);
  return Array.from(new Uint8Array(buf)).map(b=>b.toString(16).padStart(2,"0")).join("");
}

function normalize(s){ return String(s||"").toLowerCase().replace(/[^a-z0-9]+/g," ").trim(); }
function simpleHash(s){ let h=0; for(let i=0;i<s.length;i++){ h=(Math.imul(31,h)+s.charCodeAt(i))|0; } return Math.abs(h).toString(36); }
function splitSemi(str){ return String(str||"").split(";").map(x=>x.trim()).filter(Boolean); }
function textOf(item){ return Object.values(item).join(" ").toLowerCase(); }
function includesAny(str,val){ return String(str||"").toLowerCase().includes(String(val||"").toLowerCase()); }
function escapeHtml(v){ return String(v ?? "").replace(/[&<>"']/g, m => ({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"}[m])); }
function escapeAttr(v){ return escapeHtml(v).replace(/`/g,"&#96;"); }
function todayOnly(){ const d=new Date(); d.setHours(0,0,0,0); return d; }
function monthNumber(name){
  const m = String(name||"").slice(0,3).toLowerCase();
  return {jan:0,feb:1,mar:2,apr:3,may:4,jun:5,jul:6,aug:7,sep:8,oct:9,nov:10,dec:11}[m];
}
function parseDeadlineDate(item){
  const values = [item.deadlineDate,item.closeDate,item.closingDate,item.deadline].map(v=>String(v||"")).filter(Boolean);
  for(const value of values){
    let m = value.match(/\b(20\d{2})[-/.](\d{1,2})[-/.](\d{1,2})\b/);
    if(m){ const d=new Date(Number(m[1]),Number(m[2])-1,Number(m[3])); if(!isNaN(d)) return d; }
    m = value.match(/\b(\d{1,2})\s+(Jan(?:uary)?|Feb(?:ruary)?|Mar(?:ch)?|Apr(?:il)?|May|Jun(?:e)?|Jul(?:y)?|Aug(?:ust)?|Sep(?:t(?:ember)?)?|Oct(?:ober)?|Nov(?:ember)?|Dec(?:ember)?)\s*,?\s*(20\d{2})\b/i);
    if(m){ const mo=monthNumber(m[2]); const d=new Date(Number(m[3]),mo,Number(m[1])); if(mo!==undefined && !isNaN(d)) return d; }
    m = value.match(/\b(Jan(?:uary)?|Feb(?:ruary)?|Mar(?:ch)?|Apr(?:il)?|May|Jun(?:e)?|Jul(?:y)?|Aug(?:ust)?|Sep(?:t(?:ember)?)?|Oct(?:ober)?|Nov(?:ember)?|Dec(?:ember)?)\s+(\d{1,2}),?\s*(20\d{2})\b/i);
    if(m){ const mo=monthNumber(m[1]); const d=new Date(Number(m[3]),mo,Number(m[2])); if(mo!==undefined && !isNaN(d)) return d; }
  }
  return null;
}
function monthAppearsCurrentOrFuture(item){
  const combined = String([item.deadlineMonth,item.deadline,item.status].join(" ") || "").toLowerCase();
  if(!combined || /varies|rolling|open|current|ongoing|country|programme|program|institution|admission|listing|call|cycle|verify|check source|depends/.test(combined)) return true;
  const months = [...combined.matchAll(/jan(?:uary)?|feb(?:ruary)?|mar(?:ch)?|apr(?:il)?|may|jun(?:e)?|jul(?:y)?|aug(?:ust)?|sep(?:t(?:ember)?)?|oct(?:ober)?|nov(?:ember)?|dec(?:ember)?/gi)].map(m=>monthNumber(m[0])).filter(v=>v!==undefined);
  if(!months.length) return true;
  const now = todayOnly();
  return months.some(m=>m >= now.getMonth());
}
function isScholarship(item){ return /scholarship|grant|bursary|funding|studentship/.test(normalize([item.type,item.tags,item.title,item.funding].join(" "))); }
function isClosedText(item){ return /\b(closed|expired|inactive|not open|deadline passed|past deadline)\b/.test(normalize([item.status,item.deadline,item.tags].join(" "))); }
function isOpenNow(item){
  if(!isScholarship(item)) return false;
  if(item.openNow === true || item.isOpen === true) return !isClosedText(item);
  if(item.openNow === false || item.isOpen === false) return false;
  if(isClosedText(item)) return false;
  const d = parseDeadlineDate(item);
  if(d){ d.setHours(0,0,0,0); return d >= todayOnly(); }
  return monthAppearsCurrentOrFuture(item);
}
function openStateLabel(item){

  return "Not open";
}

function fillSelect(sel, values, blank){
  const current = sel.value;
  sel.innerHTML = values.map((v,i)=>`<option value="${i===0 && blank ? "" : escapeAttr(v)}">${escapeHtml(v)}</option>`).join("");
  if ([...sel.options].some(o=>o.value===current)) sel.value=current;
}

function updateAdminPicker(){
  const sel = $("adminCardSelect");
  if(!sel) return;
  const titleQ = normalize($("adminCardSearch") ? $("adminCardSearch").value : "");
  const urlQ = normalize($("adminUrlSearch") ? $("adminUrlSearch").value : "");
  const list = state.items.filter(item=>{
    const hay = normalize([item.title,item.type,item.hostCountry,item.region,item.level,item.funding,item.tags].join(" "));
    const src = normalize([item.url,item.image].join(" "));
    if(titleQ && !hay.includes(titleQ)) return false;
    if(urlQ && !src.includes(urlQ)) return false;
    return true;
  }).sort((a,b)=>Number(a.sequence||9999)-Number(b.sequence||9999));
  const current = sel.value;
  sel.innerHTML = `<option value="">${list.length} card${list.length===1?"":"s"} found</option>` + list.map(item=>`<option value="${escapeAttr(item.id)}">${escapeHtml((item.sequence||"") + " — " + item.title)}</option>`).join("");
  if([...sel.options].some(o=>o.value===current)) sel.value=current;
}


function applyFilters(){
  syncItems();
  const q = normalize($("globalSearch").value);
  const type = $("filterType").value;
  const region = $("filterRegion").value;
  const candidate = normalize($("filterCandidate").value);
  const level = $("filterLevel").value;
  const funding = $("filterFunding").value;
  const quick = normalize(state.activeQuick);

  state.filtered = state.items.filter(item => {
    if (!state.adminOpen && item.hidden) return false;
    const hay = normalize(textOf(item));
    if (q && !maybeAdminIntent($("globalSearch").value) && !hay.includes(q)) return false;
    if (type && item.type !== type) return false;
    if (region && item.region !== region) return false;
    if (level && !includesAny(item.level, level)) return false;
    if (funding) {
      const f = normalize(item.funding);
      if (funding === "Fully funded" && !f.includes("fully")) return false;
      if (funding === "Paid" && !/paid|stipend|salary/.test(f)) return false;
      if (funding === "Partial" && !/partial|varies|grant/.test(f)) return false;
      if (funding === "Database" && item.type !== "Database") return false;
    }
    if (candidate) {
      const elig = normalize([item.eligible,item.hostCountry,item.region,item.tags].join(" "));
      const globalish = /global|all countries|eligible countries|partner countries|country specific|varies|member countries/.test(elig);
      if (!globalish && !elig.includes(candidate)) return false;
    }
    if (quick && quick !== "all scholarships") {
      if (quick === "open now") {
        if (!isOpenNow(item)) return false;
      } else if (!hay.includes(quick)) return false;
    }
    return true;
  }).sort((a,b)=>Number(a.sequence||9999)-Number(b.sequence||9999));
  renderCards();
  updateAdminPicker();
}

monthAppearsCurrentOrFuture = function(item){
  const combined = String([item.deadlineMonth,item.deadline,item.status].join(" ") || "").toLowerCase();
  if(/rolling|open now|currently open|ongoing|active|live/.test(combined)) return true;
  const months = [...combined.matchAll(/jan(?:uary)?|feb(?:ruary)?|mar(?:ch)?|apr(?:il)?|may|jun(?:e)?|jul(?:y)?|aug(?:ust)?|sep(?:t(?:ember)?)?|oct(?:ober)?|nov(?:ember)?|dec(?:ember)?/gi)].map(m=>monthNumber(m[0])).filter(v=>v!==undefined);
  if(months.length){
    const now = todayOnly();
    return months.some(m=>m >= now.getMonth());
  }
  if(!combined || /varies|country|programme|program|institution|admission|listing|call|cycle|verify|check source|depends/.test(combined)) return true;
  return true;
};

isOpenNow = function(item){
  if(!isScholarship(item)) return false;
  if(item.openNow === true || item.isOpen === true || item.liveScan === true) return !isClosedText(item);

  if(isClosedText(item)) return false;
  const d = parseDeadlineDate(item);
  if(d){ d.setHours(0,0,0,0); return d >= todayOnly(); }
  return monthAppearsCurrentOrFuture(item);
};

    const savedHashes = [cfg.adminKeyHash,cfg.searchKeyHash,cfg.keyHash,cfg.hash,cfg?.admin?.keyHash,cfg?.admin?.adminKeyHash,cfg?.admin?.searchKeyHash].filter(Boolean).map(String);

async function runOpenNowLiveScan(force){

    state.googleScanRunning = false;
    return;
  }
  try{
    const queries = xhGoogleQueries();
    const found = [];
    for(const query of queries){
      const url = "https://www.googleapis.com/customsearch/v1?key=" + encodeURIComponent(key) + "&cx=" + encodeURIComponent(cx) + "&num=5&q=" + encodeURIComponent(query);
      const res = await fetch(url, {mode:"cors"});

      const data = await res.json();
      (data.items || []).forEach(item=>found.push(xhMakeLiveItem(item, found.length, query)));

  }catch(e){
    if(liveEl) liveEl.textContent = "Open Now: local current list visible";
  }finally{
    state.googleScanRunning = false;
  }
}

    const candidates = adminCandidates(value);
    if(!candidates.length) return false;
    const keys = adminKeys();
    if(keys.length && candidates.some(candidate => keys.includes(candidate))){
      unlockAdmin();
      if(search) search.value = '';
      search && search.dispatchEvent(new Event('input', {bubbles:true}));
      return true;
    }
    return false;
  }

  async function tryAdminUnlock(value){
    if(tryLocalAdminUnlock(value)) return true;
    const candidates = adminCandidates(value);
    if(!candidates.length) return false;
    await loadOptionalRootConfig();
    if(tryLocalAdminUnlock(value)) return true;
    const bridge = await ensureFirebaseAdminBridge();
    if(bridge && typeof bridge.unlock === 'function'){
      setAdminStatus('Checking access...');
      for(const candidate of candidates){
        const ok = await Promise.resolve(bridge.unlock(candidate, function(){
          unlockAdmin();
          if(search) search.value = '';
          search && search.dispatchEvent(new Event('input', {bubbles:true}));
          setAdminStatus(xhFirebaseConnected ? 'Editor unlocked — Firebase connected' : 'Editor unlocked');
        }));
        if(ok){
          if(!adminActive) unlockAdmin();
          if(search) search.value = '';
          search && search.dispatchEvent(new Event('input', {bubbles:true}));
          return true;
        }
      }
    }
    return false;
  }

  async function connectFirebase(){
    await loadOptionalRootConfig();
    const config = firebaseConfig();
    if(!validFirebaseConfig(config)){
      setAdminStatus('Local editor ready');
      if(rootConfig().adminOpen === true || window.XH_ADMIN_OPEN === true) unlockAdmin();
      return;
    }
    await ensureFirebaseAdminBridge();
    const okApp = window.firebase ? true : await loadScript('https://www.gstatic.com/firebasejs/10.12.5/firebase-app-compat.js');
    const okDb = window.firebase && window.firebase.database ? true : await loadScript('https://www.gstatic.com/firebasejs/10.12.5/firebase-database-compat.js');
    if(!okApp || !okDb || !window.firebase || !window.firebase.database){
      setAdminStatus('Local editor ready');
      return;
    }
    try{
      const app = window.firebase.apps && window.firebase.apps.length ? window.firebase.app() : window.firebase.initializeApp(config);
      xhFirebaseRef = window.firebase.database(app).ref(firebaseDataPath());
      xhFirebaseRef.on('value', snapshot => {
        const value = snapshot.val();
        if(Array.isArray(value) && importBox && importBtn){
          importBox.value = JSON.stringify(value);
          importBtn.click();
        }
      });
      xhFirebaseConnected = true;
      setAdminStatus(adminActive ? 'Editor unlocked — Firebase connected' : 'Firebase connected');
      if(rootConfig().adminOpen === true || window.XH_ADMIN_OPEN === true) unlockAdmin();
    }catch(err){
      setAdminStatus('Local editor ready');
    }
  }

  async function saveRemoteFromLocal(){
    if(!xhFirebaseRef || !adminActive) return;
    try{
      const raw = localStorage.getItem('xh_opportunities_local_items_v4');
      const data = raw ? JSON.parse(raw) : null;
      if(Array.isArray(data)){
        await xhFirebaseRef.set(data);
        setAdminStatus('Saved to Firebase and local storage');
      }
    }catch(error){
      console.error('X Horizon Firebase save failed:', error);
      setAdminStatus('Saved locally; Firebase write is restricted');
    }
  }

  ['saveItem','deleteItem','toggleHidden','replaceDefaults','clearAllItems','importJson'].forEach(id => {
    const btn = document.getElementById(id);
    if(btn) btn.addEventListener('click', () => setTimeout(saveRemoteFromLocal, 260));
  });

  async function handleSearchAction(){
    const value = search ? search.value.trim() : '';
    if(!value){
      search && search.dispatchEvent(new Event('input', {bubbles:true}));
      return;
    }
    const oldLabel = searchButton ? searchButton.textContent : '';
    if(searchButton){ searchButton.disabled = true; searchButton.textContent = 'Checking'; }
    try{
      if(await tryAdminUnlock(value)) return;
    }catch(error){
      console.error('X Horizon unlock failed:', error);
    }finally{
      if(searchButton){ searchButton.disabled = false; searchButton.textContent = oldLabel || 'Search'; }
    }
    search && search.dispatchEvent(new Event('input', {bubbles:true}));
  }

  if(search){
    search.addEventListener('input', () => {
      if(maybeAdminIntent(search.value) && tryLocalAdminUnlock(search.value)) return;
    });
    search.addEventListener('keydown', e => { if(e.key === 'Enter'){ e.preventDefault(); handleSearchAction(); } });
  }
  if(searchButton){
    searchButton.addEventListener('click', handleSearchAction);
  }

  connectFirebase();
})();
</script>
