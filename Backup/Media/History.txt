  const STORAGE_KEY = 'xhTimelineFullAdminStateV2';
  let timelineData = mergeSavedWithDefaults(loadFullState() || migrateOldSavedData(structuredCloneSafe(initialTimelineData)));

  function structuredCloneSafe(value){
    try{ return structuredClone(value); }catch(error){ return JSON.parse(JSON.stringify(value)); }
  }

  function readSavedList(key){
    try{
      const data = JSON.parse(localStorage.getItem(key) || '[]');
      return Array.isArray(data) ? data : [];
    }catch(error){return [];}
  }

  function readSavedObject(key){
    try{
      const data = JSON.parse(localStorage.getItem(key) || '{}');
      return data && typeof data === 'object' && !Array.isArray(data) ? data : {};
    }catch(error){return {};}
  }

  function loadFullState(){
    try{
      const saved = JSON.parse(localStorage.getItem(STORAGE_KEY) || 'null');
      return Array.isArray(saved) ? saved : null;
    }catch(error){return null;}
  }

  function migrateOldSavedData(base){
    const savedRows = readSavedList('xhTimelineExtraRows');
    savedRows.forEach(row => {
      if(row && row.era && !base.some(item => item.era === row.era)){
        base.push({id:row.id || '', era:row.era, label:row.label || 'New Timeline Line', events:Array.isArray(row.events) ? row.events : []});
      }
    });
    const savedCards = readSavedList('xhTimelineExtraCards');
    savedCards.forEach(card => {
      const row = base.find(item => item.era === card.era);
      if(row) row.events.push(card);
    });
    const savedPopEdits = readSavedObject('xhTimelinePopEdits');
    Object.keys(savedPopEdits).forEach(oldKey => {
      const parts = decodeURIComponent(oldKey).split('||');
      const row = base.find(item => item.era === parts[0]);
      if(!row) return;
      const title = parts.slice(2).join('||');
      const event = row.events.find(item => String(item.date || '') === parts[1] && String(item.title || '') === title);
      if(event) Object.assign(event, savedPopEdits[oldKey]);
    });
    return base;
  }


  function mergeSavedWithDefaults(savedValue){
    const defaults = structuredCloneSafe(initialTimelineData);
    const saved = Array.isArray(savedValue) ? structuredCloneSafe(savedValue) : [];
    const merged = [];
    const usedEras = new Set();
    const clean = value => String(value || '').trim().toLowerCase();
    const eventKey = event => clean(event.date) + '||' + clean(event.title);

    defaults.forEach(defaultRow => {
      const savedRow = saved.find(row => row && clean(row.era) === clean(defaultRow.era));
      const row = savedRow ? Object.assign({}, defaultRow, savedRow) : structuredCloneSafe(defaultRow);
      const events = Array.isArray(row.events) ? row.events : [];
      const existing = new Map();
      events.forEach(event => {
        if(event && typeof event === 'object') existing.set(eventKey(event), event);
      });
      (Array.isArray(defaultRow.events) ? defaultRow.events : []).forEach(defaultEvent => {
        const key = eventKey(defaultEvent);
        const current = existing.get(key);
        if(current){
          Object.keys(defaultEvent).forEach(field => {
            if(current[field] === undefined || current[field] === null || current[field] === '') current[field] = structuredCloneSafe(defaultEvent[field]);
          });
        }else{
          events.push(structuredCloneSafe(defaultEvent));
        }
      });
      row.events = events;
      merged.push(row);
      usedEras.add(clean(row.era));
    });

    saved.forEach(row => {
      if(row && row.era && !usedEras.has(clean(row.era))) merged.push(row);
    });
    return merged;
  }

  function saveFullState(){
    try{ localStorage.setItem(STORAGE_KEY, JSON.stringify(timelineData)); }catch(error){ console.warn('XH state save failed', error); }
  }

  function normalizeIds(){
    let used = new Set();
    let nextId = 1001;
    timelineData.forEach((row, rowIndex) => {
      if(!row.id) row.id = 'line-' + (rowIndex + 1);
      if(!Array.isArray(row.events)) row.events = [];
      row.events.forEach(event => {
        let id = String(event.id || event.cardId || '').replace(/\D/g, '');
        while(!id || used.has(id)){ id = String(nextId++); }
        event.id = id;
        event.cardId = id;
        used.add(id);
        const numeric = Number(id);
        if(Number.isFinite(numeric) && numeric >= nextId) nextId = numeric + 1;
        event.type = event.type || 'Mission';
        event.status = event.status || 'Status';
        event.fx = event.fx || 'fx-aurora';
        event.image = event.image || FALLBACK_IMG;
        event.desc = event.desc || '';
      });
    });
  }

  function renderAdminSelectors(){

    const previousPop = document.getElementById('popTargetCard')?.value || '';
    const options = [];
    timelineData.forEach(row => row.events.forEach(event => {
      options.push(`<option value="${safeText(event.id)}">ID ${safeText(event.id)} — ${safeText(row.era)} — ${safeText(event.date)} — ${safeText(event.title)}</option>`);
    }));
    const cardSelect = document.getElementById('cardEditSelect');
    const popSelect = document.getElementById('popTargetCard');
    if(cardSelect){
      cardSelect.innerHTML = options.join('') || '<option value="">No cards available</option>';

  function wireCards(){
    document.querySelectorAll('.xh-event-card').forEach(card => {
      const open = () => openPopupById(card.dataset.cardId);
      card.addEventListener('click', open);
      card.addEventListener('keydown', e => { if(e.key === 'Enter' || e.key === ' '){ e.preventDefault(); open(); } });
      card.addEventListener('pointermove', e => {
        if(window.matchMedia && window.matchMedia('(pointer: coarse)').matches) return;
        const rect = card.getBoundingClientRect();
        const x = (e.clientX - rect.left) / rect.width - 0.5;
        const y = (e.clientY - rect.top) / rect.height - 0.5;
        card.style.setProperty('--xh-rx', `${(-y * 4).toFixed(2)}deg`);
        card.style.setProperty('--xh-ry', `${(x * 4).toFixed(2)}deg`);
        card.classList.add('is-card-active');
      });
      card.addEventListener('pointerleave', () => {
        card.style.setProperty('--xh-rx', '0deg');
        card.style.setProperty('--xh-ry', '0deg');
        card.classList.remove('is-card-active');
      });
    });
  }

  function wireCollapse(){
    document.querySelectorAll('[data-collapse]').forEach(btn => {
      btn.addEventListener('click', () => {
        const row = document.getElementById(btn.dataset.collapse);
        row.classList.toggle('is-hidden');
        btn.textContent = row.classList.contains('is-hidden') ? 'Show' : 'Hide';
      });
    });
  }

  function setupSearch(){
    const input = document.getElementById('xhSearchBox');
    const searchBtn = document.getElementById('xhSearchButton');
    const results = document.getElementById('xhSearchResults');
    const adminPanel = document.getElementById('xhAdminPanel');
    if(!input || !searchBtn) return;

    function enableAdminFromSearch(){
      if(adminPanel) adminPanel.classList.add('is-open');
      document.body.setAttribute('data-xh-admin','unlocked');
      document.body.classList.add('editor-on');
      input.value = '';
      input.placeholder = 'Admin unlocked';
      if(results){ results.classList.remove('show'); results.innerHTML = ''; }
      document.querySelectorAll('.xh-event-card').forEach(card => card.classList.remove('xh-filter-hidden'));
      document.querySelectorAll('.xh-decade-row').forEach(row => row.classList.remove('xh-filter-hidden'));
      setTimeout(() => adminPanel?.scrollIntoView({behavior:'smooth', block:'start'}), 80);
    }

    window.xhUnlockHiddenAdmin = enableAdminFromSearch;

    function waitForUnlockSystem(){
      return new Promise(resolve => {
        if(window.XHorizonAdmin && typeof window.XHorizonAdmin.unlock === 'function') return resolve(window.XHorizonAdmin);
        let tries = 0;
        const timer = setInterval(() => {
          tries++;
          if(window.XHorizonAdmin && typeof window.XHorizonAdmin.unlock === 'function'){
            clearInterval(timer);
            resolve(window.XHorizonAdmin);
          }else if(tries > 12){
            clearInterval(timer);
            resolve(null);
          }
        }, 120);
      });
    }

    function filterTimeline(value){
      const q = String(value || '').trim().toLowerCase();
      let matches = [];
      document.querySelectorAll('.xh-decade-row').forEach(row => {
        let visibleCount = 0;
        row.querySelectorAll('.xh-event-card').forEach(card => {
          const text = [card.dataset.cardId, card.dataset.title, card.dataset.date, card.dataset.type, card.dataset.status, card.dataset.desc, card.dataset.popBody, card.dataset.popHeadings, row.dataset.era].join(' ').toLowerCase();
          const match = !q || text.includes(q);
          card.classList.toggle('xh-filter-hidden', !match);
          if(match){
            visibleCount++;
            if(q && matches.length < 8) matches.push(card);
          }
        });
        row.classList.toggle('xh-filter-hidden', !visibleCount);
      });

      if(!results) return;
      results.innerHTML = '';
      if(!q){ results.classList.remove('show'); return; }
      if(matches.length){
        matches.forEach(card => {
          const item = document.createElement('button');
          item.type = 'button';
          item.className = 'xh-result-item';
          item.innerHTML = '<strong>' + safeText(card.dataset.title || 'Timeline event') + '</strong><span>' + safeText((card.dataset.date || '') + ' · ' + (card.dataset.type || '')) + '</span>';
          item.addEventListener('click', () => {
            results.classList.remove('show');
            card.scrollIntoView({behavior:'smooth', block:'center'});
            setTimeout(() => openPopupById(card.dataset.cardId), 220);
          });
          results.appendChild(item);
        });
      }else{
        results.innerHTML = '<div class="xh-result-item"><strong>No result</strong><span>Try another timeline keyword.</span></div>';
      }
      results.classList.add('show');
    }

    async function handleSearch(){
      const value = input.value.trim();
      if(!value){ filterTimeline(''); return; }
      searchBtn.disabled = true;
      const oldLabel = searchBtn.textContent;
      searchBtn.textContent = 'Checking';
      try{
        const bridge = await waitForUnlockSystem();
        if(bridge && typeof bridge.unlock === 'function'){
          let callbackDone = false;
          const ok = await Promise.resolve(bridge.unlock(value, () => { callbackDone = true; enableAdminFromSearch(); }));
          if(ok){
            if(!callbackDone) enableAdminFromSearch();
            return;
          }
        }
      }catch(error){
        console.error('X Horizon unlock failed:', error);
      }finally{
        searchBtn.disabled = false;
        searchBtn.textContent = oldLabel;
      }
      filterTimeline(value);
    }

    input.addEventListener('input', () => filterTimeline(input.value));
    input.addEventListener('keydown', e => { if(e.key === 'Enter'){ e.preventDefault(); handleSearch(); } });
    searchBtn.addEventListener('click', handleSearch);
    document.addEventListener('click', e => { if(results && !e.target.closest('.xh-search-wrap')) results.classList.remove('show'); });
  }

  function setupTabs(){
    tabsEl.addEventListener('click', e => {
      const tab = e.target.closest('.xh-tab');
      if(!tab) return;
      tabsEl.querySelectorAll('.xh-tab').forEach(item => item.classList.remove('is-active'));
      tab.classList.add('is-active');
      document.getElementById(tab.dataset.jump)?.scrollIntoView({behavior:'smooth', block:'start'});
    });
  }

  document.getElementById('xhPopClose')?.addEventListener('click', closePopup);
  document.getElementById('xhPopBackdrop')?.addEventListener('click', e => { if(e.target.id === 'xhPopBackdrop') closePopup(); });
  document.addEventListener('keydown', e => { if(e.key === 'Escape') closePopup(); });

  renderTabs();
  renderRows();
  renderWavePreview();
  setupAdmin();
  setupSearch();
  setupTabs();
})();
