<script type="module">
import { initializeApp, getApps } from "https://www.gstatic.com/firebasejs/10.12.5/firebase-app.js";
import { getAuth, signInWithCustomToken, onAuthStateChanged } from "https://www.gstatic.com/firebasejs/10.12.5/firebase-auth.js";
import { getFunctions, httpsCallable } from "https://www.gstatic.com/firebasejs/10.12.5/firebase-functions.js";

const fallbackFirebaseConfig = {
  apiKey: "YOUR_FIREBASE_API_KEY",
  authDomain: "YOUR_PROJECT.firebaseapp.com",
  projectId: "YOUR_PROJECT_ID",
  storageBucket: "YOUR_PROJECT.appspot.com",
  messagingSenderId: "YOUR_SENDER_ID",
  appId: "YOUR_APP_ID"
};

const firebaseConfig = window.XHORIZON_FIREBASE_CONFIG || window.XHorizonFirebaseConfig || fallbackFirebaseConfig;
function configured(cfg){
  return cfg &&
    cfg.apiKey && !String(cfg.apiKey).includes("YOUR_") &&
    cfg.projectId && !String(cfg.projectId).includes("YOUR_");
}

let app = null;
let auth = null;
let functions = null;

try {
  if (configured(firebaseConfig)) {
    app = getApps().length ? getApps()[0] : initializeApp(firebaseConfig);
    auth = getAuth(app);
    functions = getFunctions(app);
  }
} catch (error) {
  console.error("Firebase initialization failed:", error);
}

if (configured(firebaseConfig) && !window.XHorizonAdmin) {
  window.XHorizonAdmin = {
    async unlock(key, done) {
      const cleanKey = String(key || "").trim();
      if (!cleanKey || !auth || !functions) return false;
      try {
        const verifyAdminKey = httpsCallable(functions, "verifyAdminKey");
        const result = await verifyAdminKey({ key: cleanKey });
        const data = result && result.data ? result.data : {};
        const customToken = data.token || data.customToken || data.firebaseToken || "";
        if (customToken) await signInWithCustomToken(auth, customToken);

        const user = auth.currentUser;
        if (!user) return false;

        const tokenResult = await user.getIdTokenResult(true);
        const allowed = tokenResult && tokenResult.claims && tokenResult.claims.admin === true;
        if (allowed) {
          document.body.classList.add("editor-on");
          if (typeof done === "function") done();
          return true;
        }
        return false;
      } catch (error) {
        console.error("X Horizon unlock failed:", error);
        return false;
      }
    }
  };
}

if (auth) {
  onAuthStateChanged(auth, async function(user) {
    if (!user) return;
    try {
      const tokenResult = await user.getIdTokenResult(true);
      if (tokenResult && tokenResult.claims && tokenResult.claims.admin === true) {
        document.body.classList.add("editor-on");
      }
    } catch (error) {
      console.error("State check failed:", error);
    }
  });
}
</script>

        var str = String(value || '');

        function moveDrag(e){

            if (drag.handle === 'bottom') f.bottom = drag.frame.bottom - dy;
            if (drag.handle === 'left') f.left = drag.frame.left + dx;
            if (drag.handle === 'right') f.right = drag.frame.right - dx;
            if (drag.handle === 'move') { f.x = drag.frame.x + dx; f.y = drag.frame.y + dy; }
            setFrameInputs(normalizeFrame(f));
            updateVisualPreview();
        }
        function endDrag(){ drag = null; }
        previewShell.addEventListener('pointerdown', startDrag);
        window.addEventListener('pointermove', moveDrag);
        window.addEventListener('pointerup', endDrag);
        window.addEventListener('pointercancel', endDrag);
    }

    function withYouTubeOptions(url){
        var src = String(url || '');
        if (src.indexOf('youtube.com/embed/') === -1) return src;
        var joiner = src.indexOf('?') === -1 ? '?' : '&';
        var options = [];
        if (src.indexOf('rel=') === -1) options.push('rel=0');
        if (src.indexOf('modestbranding=') === -1) options.push('modestbranding=1');
        if (src.indexOf('playsinline=') === -1) options.push('playsinline=1');
        return options.length ? src + joiner + options.join('&') : src;
    }

    function instagramEmbedUrl(kind, code){
        var cleanKind = String(kind || 'reel').toLowerCase();
        var cleanCode = String(code || '').replace(/[^a-zA-Z0-9_-]/g,'');
        return 'https://www.instagram.com/' + cleanKind + '/' + cleanCode + '/embed/?utm_source=ig_embed';
    }

    function mediaEmbed(value){
        var raw = String(value || '').trim();
        if (!raw) return null;

        if (/^[a-zA-Z0-9_-]{8,}$/.test(raw) && raw.indexOf('http') !== 0) {
            return { embed:withYouTubeOptions('https://www.youtube.com/embed/' + raw), provider:'youtube', label:'YouTube Video' };
        }

        var ytPatterns = [
            /youtu\.be\/([a-zA-Z0-9_-]{8,})/,
            /youtube\.com\/watch\?v=([a-zA-Z0-9_-]{8,})/,
            /youtube\.com\/embed\/([a-zA-Z0-9_-]{8,})/,
            /youtube\.com\/shorts\/([a-zA-Z0-9_-]{8,})/,
            /[?&]v=([a-zA-Z0-9_-]{8,})/
        ];
        for (var i = 0; i < ytPatterns.length; i++) {
            var yt = raw.match(ytPatterns[i]);
            if (yt && yt[1]) return { embed:withYouTubeOptions('https://www.youtube.com/embed/' + yt[1]), provider:'youtube', label:'YouTube Video' };
        }

        var igEmbed = raw.match(/instagram\.com\/(p|reel|tv)\/([^\/?#]+)\/embed/i);
        if (igEmbed && igEmbed[1] && igEmbed[2]) {
            return { embed:instagramEmbedUrl(igEmbed[1].toLowerCase(), igEmbed[2]), provider:'instagram', label: igEmbed[1].toLowerCase() === 'reel' ? 'Instagram Reel' : 'Instagram Video' };
        }
        var ig = raw.match(/instagram\.com\/(p|reel|tv)\/([^\/?#]+)/i);
        if (ig && ig[1] && ig[2]) {
            return { embed:instagramEmbedUrl(ig[1].toLowerCase(), ig[2]), provider:'instagram', label: ig[1].toLowerCase() === 'reel' ? 'Instagram Reel' : 'Instagram Video' };
        }

        return { embed:raw, provider:'custom', label:'Video' };
    }

    function cleanTitle(value, media){
        var title = safeText(value);
        if (!title || /https?:\/\//i.test(title) || /instagram\.com|youtu\.be|youtube\.com/i.test(title)) {
            title = media && media.label ? media.label : 'X Horizon Video';
        }
        title = title.replace(/\b(reel|p|tv)\/[a-zA-Z0-9_-]+\b/gi, '').replace(/\s{2,}/g,' ').trim();
        return title || 'X Horizon Video';
    }

    function configureIframe(iframe, provider, src){
        var cleanProvider = provider === 'instagram' ? 'instagram' : (provider === 'custom' ? 'custom' : 'youtube');
        iframe.src = cleanProvider === 'youtube' ? withYouTubeOptions(src) : src;
        iframe.allowFullscreen = true;
        iframe.setAttribute('allowfullscreen','allowfullscreen');
        iframe.setAttribute('webkitallowfullscreen','webkitallowfullscreen');
        iframe.setAttribute('mozallowfullscreen','mozallowfullscreen');
        iframe.setAttribute('scrolling','no');
        iframe.loading = 'lazy';
        iframe.referrerPolicy = 'strict-origin-when-cross-origin';
        iframe.allow = 'accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share; fullscreen';
        iframe.classList.remove('xh-ig-frame','xh-yt-frame');
        if (cleanProvider === 'instagram') {
            iframe.classList.add('xh-ig-frame');
            iframe.removeAttribute('sandbox');
            iframe.setAttribute('allow','autoplay; clipboard-write; encrypted-media; picture-in-picture; web-share; fullscreen');
        } else {
            iframe.classList.add('xh-yt-frame');
            iframe.removeAttribute('sandbox');
        }
    }

    function waitForUnlockSystem(){
        return new Promise(function(resolve){
            if (window.XHorizonAdmin && typeof window.XHorizonAdmin.unlock === 'function') return resolve(window.XHorizonAdmin);
            var tries = 0;
            var timer = setInterval(function(){
                tries++;
                if (window.XHorizonAdmin && typeof window.XHorizonAdmin.unlock === 'function') {
                    clearInterval(timer);
                    resolve(window.XHorizonAdmin);
                } else if (tries > 12) {
                    clearInterval(timer);
                    resolve(null);
                }
            }, 120);
        });
    }

    async function handleSearchAction(){
        var value = input ? input.value.trim() : '';
        if (!value) {
            runSearch('');
            return;
        }

        var oldLabel = searchBtn ? searchBtn.textContent : '';
        if (searchBtn) {
            searchBtn.disabled = true;
            searchBtn.textContent = 'Checking';
        }

        try {
            var bridge = await waitForUnlockSystem();
            if (bridge && typeof bridge.unlock === 'function') {
                var callbackDone = false;
                var ok = await Promise.resolve(bridge.unlock(value, function(){
                    callbackDone = true;
                    unlockAdmin();
                }));
                if (ok) {
                    if (!callbackDone) unlockAdmin();
                    return;
                }
            }
        } catch (error) {
            console.error('X Horizon video admin unlock failed:', error);
        } finally {
            if (searchBtn) {
                searchBtn.disabled = false;
                searchBtn.textContent = oldLabel || 'Search';
            }
        }

        runSearch(value);
    }

    function runSearch(value){
        var query = normalizeText(value);
        removeEmptyNotes();
        if (!query) {
            restoreDefault();
            installDeleteButtons();
            return;
        }
        var total = 0;
        getSections().forEach(function(sec){
            var matchInSection = 0;
            sec.style.display = 'block';
            sec.querySelectorAll('.vid-card').forEach(function(card){
                if (card.getAttribute('data-xh-hidden') === 'true') {
                    card.style.display = 'none';
                    return;
                }
                var title = card.querySelector('.vid-title');
                var text = normalizeText(title ? title.textContent : '');
                var matched = text.indexOf(query) !== -1;
                card.style.display = matched ? '' : 'none';
                if (matched) { matchInSection++; total++; }
            });
            if (!matchInSection) sec.style.display = 'none';
        });
        if (status) status.textContent = total ? (total + ' video heading match' + (total === 1 ? '' : 'es') + ' found.') : 'No video heading matched your search.';
        installDeleteButtons();
        setTimeout(syncCategoryVideoPlayback, 0);
    }

        var row = document.getElementById(rowId);
        if (!row) return;
        row.querySelectorAll('div[onclick]').forEach(function(card){
            card.addEventListener('click', function(){
                if (input) input.value = '';
                removeEmptyNotes();
                getSections().forEach(function(sec){
                    sec.querySelectorAll('.vid-card').forEach(function(videoCard){
                        if (videoCard.getAttribute('data-xh-hidden') === 'true') videoCard.style.display = 'none';
                        else videoCard.style.display = '';
                    });
                });
                if (status) status.textContent = 'Search video headings here.';
                installDeleteButtons();
                setTimeout(syncCategoryVideoPlayback, 0);
            });
        });
    });

    if (lockBtn) lockBtn.addEventListener('click', lockAdmin);

    if (addBtn) addBtn.addEventListener('click', function(){
        var media = mediaEmbed(urlInput ? urlInput.value : '');
        var title = cleanTitle(titleInput ? titleInput.value : '', media);
        var category = catInput ? catInput.value : 'video-content';
        if (!media || !media.embed) {
            if (adminMsg) adminMsg.textContent = 'Add a valid YouTube or Instagram video URL.';
            return;
        }
        var items = readItems();
        items.push({ id:'xhv_' + Date.now(), title:title, embed:media.embed, provider:media.provider, category:category });
        writeItems(items);
        if (titleInput) titleInput.value = '';
        if (urlInput) urlInput.value = '';
        renderAdminVideos();
        if (adminMsg) adminMsg.textContent = 'Video added successfully inside selected category.';
        runSearch(input ? input.value : '');
    });

    clearLegacyVideosOnce();
    seedInstagramTestVideos();
    prepareStaticCards();
    renderAdminVideos();
})();
