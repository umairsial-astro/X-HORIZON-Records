function normalise(json) {

  rows.forEach(function (a) {

    if (!url || !title || !src || isNaN(pub) || seen[title.toLowerCase()]) return;

function normaliseRss(feedName, json) {
  var rows = (json && json.items) || [], out = [];
  rows.forEach(function (a) {
    var url = safeUrl(a.link), title = decode(a.title || '').trim();
    /* RSS dates commonly already contain GMT/UTC or an explicit offset; do not append a second Z. */
    var rawDate = String(a.pubDate || a.published || a.updated || '').trim();
    var pub = Date.parse(rawDate);
    if (isNaN(pub) && rawDate) pub = Date.parse(rawDate.replace(/\s+GMT$/i, ' UTC'));
    var sum = decode(String(a.description || a.content || '').replace(/<[^>]*>/g, ' ')).replace(/\s+/g, ' ').trim();
    if (sum.length > 280) sum = sum.slice(0, 277).replace(/\s+\S*$/, '') + '\u2026';

function helio(e, T) {
  var a = e[0] + e[1] * T, ec = e[2] + e[3] * T, I = (e[4] + e[5] * T) * D2R, L = e[6] + e[7] * T, w = e[8] + e[9] * T, O = (e[10] + e[11] * T) * D2R;
  var M = (((L - w) % 360) + 540) % 360 - 180; M *= D2R;
  var E = M; for (var i = 0; i < 10; i++) E = M + ec * Math.sin(E);
  var xv = a * (Math.cos(E) - ec), yv = a * Math.sqrt(1 - ec * ec) * Math.sin(E);
  var v = Math.atan2(yv, xv), r = Math.sqrt(xv * xv + yv * yv), u = v + (w * D2R - O);
  return [r * (Math.cos(O) * Math.cos(u) - Math.sin(O) * Math.sin(u) * Math.cos(I)), r * (Math.sin(O) * Math.cos(u) + Math.cos(O) * Math.sin(u) * Math.cos(I)), r * Math.sin(u) * Math.sin(I)];
}
function planetsNow(ms) {

    var p = helio(PL[n], T), g = [p[0] - ea[0], p[1] - ea[1], p[2] - ea[2]], gl = Math.sqrt(g[0] * g[0] + g[1] * g[1] + g[2] * g[2]);

    var d = document.getElementById('skIssD'); d.textContent = '';
    d.appendChild(document.createTextNode(Math.round(j.altitude) + ' km up · ' + Math.round(j.velocity).toLocaleString() + ' km/h · '));
    d.appendChild(h('a', { href: C.links.satellites, text: 'pass times' }));
  }).catch(function () { setT('skIssV', 'Unavailable'); setT('skIssD', 'Position feed not reachable right now.'); });
}
function loadLaunch() {
  var c = cacheGet('launch', 20 * 60000), p = c ? Promise.resolve(c.v) : fetchJSON(C.launchUrl, 8000).then(function (j) { cacheSet('launch', j); return j; });
  return p.catch(function () { var o = cacheGet('launch', 24 * 3600e3); if (o) return o.v; throw 0; }).then(function (j) {
    var now = Date.now(), r = (j.results || []).filter(function (x) { return Date.parse(x.net) > now - 3600e3; })[0] || (j.results || [])[0];
    if (!r) throw 0;
    var parts = (r.name || 'Launch').split('|'), mission = parts.length > 1 ? parts.slice(1).join('|').trim() : parts[0].trim(), vehicle = parts.length > 1 ? parts[0].trim() + ' · ' : '';
    setT('skLaunchV', mission);
    setT('skLaunchD', vehicle + 'NET ' + fmtDT(Date.parse(r.net)) + (r.status && r.status.abbrev ? ' · ' + r.status.abbrev : ''));
  }).catch(function () { setT('skLaunchV', 'Unavailable'); setT('skLaunchD', 'Launch schedule not reachable right now.'); });
}
